# Ataque básico
sudo ./fydelisrarzip.pl -f backup.zip -w rockyou.txt

# Ataque paralelo com 8 threads + mutações
sudo ./fydelisrarzip.pl -f dados.rar -w wordlist.txt -t 8 -m

# Ataque incremental (força bruta) 1-6 chars alfanuméricos
sudo ./fydelisrarzip.pl -f arquivo.7z -i --min-len 1 --max-len 6 --charset lud

# Resume de ataque anterior
sudo ./fydelisrarzip.pl -f protegido.zip -w comum.txt --resume

# Benchmark do backend
sudo ./fydelisrarzip.pl -f teste.zip --benchmark

# Apenas as 1000 primeiras tentativas, timeout 30s
sudo ./fydelisrarzip.pl -f alvo.zip -w senhas.txt --max-attempts 1000 --timeout 30

# Exportar resultado
sudo ./fydelisrarzip.pl -f confidential.rar -w rockyou.txt -o senha_encontrada.txt

# Com regras customizadas
sudo ./fydelisrarzip.pl -f backup.zip -w common.txt -r minhas_regras.txt
