# Básico MD5
perl fydelishash.pl -h 5f4dcc3b5aa765d61d8327deb882cf99 -w senhas.txt

# Auto-detect SHA256
perl fydelishash.pl -h 9f86d081884c7d659a2feaa0c55ad015a3bf4f1b2b0b822cd15d6c15b0f00a08 -w dict.txt

# NTLM com regras hashcat-style
perl fydelishash.pl -h 209c6174da490caeb422f3fa5a7ae634 -w rockyou.txt -r rules/best64.rule -T 8

# Múltiplos hashes em lote
perl fydelishash.pl -H hashes.txt -w passwords.txt -t sha1 -o found.txt

# Com regras inline e case permutations
perl fydelishash.pl -h 5f4dcc3b5aa765d61d8327deb882cf99 -w dict.txt \
    -R 'l $1' -R 'u $!' -R 'c r $123' \
    --case-perm \
    -T 4 \
    -v

# Listar tipos suportados
perl fydelishash.pl --list-hash-types

# Salted hash + append numbers
perl fydelishash.pl -h hash_value.txt -w dict.txt --salt "meusalt" --append-num --num-range 0-9999
