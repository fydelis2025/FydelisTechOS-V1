# Básico: nome + sobrenome + ano
perl fydeliswordlist.pl -n Maria -s Silva -a 1990 -o output/maria.txt

# Com leet speak e símbolos
perl fydeliswordlist.pl -n Admin -s Server -k corp --with-leet --with-symbols -o output/admin.txt

# Apenas estimativa (não gera)
perl fydeliswordlist.pl -k empresa -w "2024,portal,vpn" --with-leet --estimate

# Completo com verbose
perl fydeliswordlist.pl \
    -n Joao \
    -s Santos \
    -a 1985 \
    -l "Rio de Janeiro" \
    -k admin \
    -w "teste,dev,backup,senha" \
    --with-leet \
    --with-symbols \
    --with-reversed \
    --with-doubled \
    --with-truncated \
    --max-words 500000 \
    -o output/joao.txt \
    -v

# Empresa: nome da empresa + palavras relacionadas
perl fydeliswordlist.pl \
    -k "EmpresaX" \
    -w "vpn,portal,rh,ti,adm,suporte,2024,2025" \
    --with-leet \
    --with-symbols \
    --max-words 200000 \
    -o output/empresax.txt
