# Básico (usa wordlist padrão interna)
perl fydelisdir.pl -h example.com

# HTTPS com wordlist personalizada e extensões
perl fydelisdir.pl \
    -h target.com \
    -s \
    -w wordlists/directories.txt \
    -e php,html,bak,txt,old \
    -T 20 \
    -t 10 \
    -o output/encontrados.txt

# Scan com filtros anti-falso-positivo
perl fydelisdir.pl \
    -h 10.0.0.1 \
    --sc "200,403,500" \
    --min-size 50 \
    --exclude-text "not found" \
    -v

# Autenticado com recursão
perl fydelisdir.pl \
    -h admin.example.com \
    -s \
    -u admin \
    -P 's3nh4!' \
    -r 2 \
    -o output/admin.txt

# Delay controlado para não sobrecarregar
perl fydelisdir.pl \
    -h 192.168.1.100 \
    -w wordlists/comum.txt \
    -T 5 \
    -d 200 \
    -q

# Apenas verbose (ver todas tentativas)
perl fydelisdir.pl \
    -h dev.internal.com \
    -e asp,aspx,config \
    -v
